  • Hubs are a fixed list of 10: the 9 category slugs from the sitemap (properties, cars-and-vehicles, occasions-and-entertainment, services, devices-and-appliances, equipment-and-tools, courts-and-arenas, furniture-and-accessories, fashion-and-clothes) plus guides. Any other [hub] value is a 404. page, author, tag are reserved and cannot be slugs.
  • Slugs are Latin, lowercase, hyphenated, identical in both locales (same rule as /ar/cars-and-vehicles/cars/). Google recommends descriptive words with hyphens and percent-encoding for non-ASCII, so Arabic slugs would ship as 100+ character encoded strings (URL structure).
  • Trailing slash on every URL, as on the rest of the site.
  • An article whose en_published is false returns 404 on /en/ and emits only ar + x-default hreflang. Never emit an hreflang to a 404 (Localized versions).
  • Every list page is server-rendered with its own URL and a self canonical; page 2 does not canonical to page 1 (Pagination). No <link rel="next/prev"> in the head; Google ignores it. No infinite scroll on the blog.
  • Slug or hub changes go through the post_redirects table and Next redirects() (301), never a silent rename.

The helper

// lib/blog/jsonld.ts
const SITE = 'https://renter.net';
const ORG_ID = `${SITE}/#organization`;   // the Organization already on the home page; never repeat the block

export function blogJsonLd(post: Post, locale: 'ar' | 'en') {
  const url = `${SITE}/${locale}/blog/${post.hub}/${post.slug}/`;
  const article = {
    '@context': 'https://schema.org',
    '@type': 'BlogPosting',
    '@id': `${url}#article`,
    mainEntityOfPage: { '@type': 'WebPage', '@id': url },
    url,
    headline: post.title,                                   // = H1, 110 chars max
    description: post.excerpt,
    image: [post.cover.w1600, post.cover.w1200x900, post.cover.w1200x1200],   // 16:9, 4:3, 1:1 (Google Article guide)
    datePublished: post.publish_at,                         // ISO 8601 with +03:00
    dateModified: post.updated_at_content,
    inLanguage: locale,
    articleSection: post.hub_name,
    wordCount: post.word_count,
    author: { '@type': 'Person', name: post.author.name, url: `${SITE}/${locale}/blog/author/${post.author.slug}/` },
    publisher: { '@id': ORG_ID },
    isPartOf: { '@id': `${SITE}/${locale}/#website` },
  };
  const breadcrumb = {
    '@context': 'https://schema.org',
    '@type': 'BreadcrumbList',
    itemListElement: [
      { '@type': 'ListItem', position: 1, name: locale === 'ar' ? 'الرئيسية' : 'Home', item: `${SITE}/${locale}/` },
      { '@type': 'ListItem', position: 2, name: locale === 'ar' ? 'المدونة' : 'Blog', item: `${SITE}/${locale}/blog/` },
      { '@type': 'ListItem', position: 3, name: post.hub_name, item: `${SITE}/${locale}/blog/${post.hub}/` },
      { '@type': 'ListItem', position: 4, name: post.title },      // last item: no "item" (Google breadcrumb doc)
    ],
  };
  return { article, breadcrumb };
}

export async function generateMetadata({ params }: { params: Promise<Params> }): Promise<Metadata> {
  const { locale, hub, slug } = await params;
  const post = await getPost(slug, locale);
  if (!post || post.hub !== hub) return {};
  const path = `/${locale}/blog/${hub}/${slug}/`;
  const languages: Record<string, string> = { ar: `${SITE}/ar/blog/${hub}/${slug}/`, 'x-default': `${SITE}/ar/blog/${hub}/${slug}/` };
  if (post.en_published) languages.en = `${SITE}/en/blog/${hub}/${slug}/`;
  return {
    title: post.meta_title,                       // "سعر إيجار السيارات الشهري في الرياض 2026 | رينتر"
    description: post.excerpt,
    robots: { index: true, follow: true },
    alternates: { canonical: `${SITE}${path}`, languages },
    openGraph: {
      type: 'article',
      title: post.meta_title,
      description: post.excerpt,
      url: `${SITE}${path}`,
      siteName: 'Renter',
      images: [{ url: post.cover.w1600, width: 1600, height: 900 }],
      publishedTime: post.publish_at,             // "2026-09-20T09:00:00+03:00"
      modifiedTime: post.updated_at_content,
      section: post.hub_name,
      tags: post.tags,
    },
    twitter: { card: 'summary_large_image', title: post.meta_title, description: post.excerpt, images: [post.cover.w1600] },
  };
}

5. Index and hub pages with pagination

One component serves four routes. 12 cards per page. Real <a href> pager. Beyond the last page is a 404; /page/1/ is a 301 to the base URL.

// app/[locale]/blog/[hub]/page.tsx  and  app/[locale]/blog/[hub]/page/[n]/page.tsx  (same component, n defaults to 1)
// app/[locale]/blog/page.tsx        and  app/[locale]/blog/page/[n]/page.tsx        (hub = undefined)
import type { Metadata } from 'next';
import { notFound, permanentRedirect } from 'next/navigation';
import { getPosts, getHub } from '@/lib/blog/api';
import { BlogCard } from '@/components/blog/BlogCard';

const SITE = 'https://renter.net';
const PER_PAGE = 12;

export const revalidate = 600;

export async function generateMetadata({ params }: { params: Promise<{ locale: 'ar'|'en'; hub?: string; n?: string }> }): Promise<Metadata> {
  const { locale, hub, n } = await params;
  const page = Number(n ?? 1);
  const base = hub ? `/${locale}/blog/${hub}/` : `/${locale}/blog/`;
  const path = page > 1 ? `${base}page/${page}/` : base;
  const h = hub ? await getHub(hub, locale) : null;
  const title = hub
    ? (locale === 'ar' ? `أدلة وأسعار ${h!.name} في السعودية | رينتر` : `${h!.name} Rental Guides and Prices in Saudi Arabia | Renter`)
    : (locale === 'ar' ? 'مدونة رينتر: أدلة وأسعار الإيجار في السعودية' : 'Renter Blog: Rental Guides and Prices in Saudi Arabia');
  const suffix = page > 1 ? (locale === 'ar' ? `، الصفحة ${page}` : `, page ${page}`) : '';
  const description = hub ? h!.intro : (locale === 'ar'
    ? 'أسعار حقيقية من مؤجرين على المنصة، شروط، ومقارنات لكل ما يمكن استئجاره: سيارات، معدات، مناسبات، عقارات وأجهزة.'
    : 'Real prices from lessors on the platform, requirements, and comparisons for everything you can rent: cars, equipment, events, property and devices.');
  return {
    title: title.replace(' | ', `${suffix} | `),
    description,
    robots: { index: true, follow: true },
    alternates: {
      canonical: `${SITE}${path}`,                                  // self canonical, also on page 2+
      languages: { ar: `${SITE}/ar${path.slice(3)}`, en: `${SITE}/en${path.slice(3)}`, 'x-default': `${SITE}/ar${path.slice(3)}` },
    },
    openGraph: { type: 'website', title, description, url: `${SITE}${path}`, siteName: 'Renter', images: [{ url: `${SITE}/renterlogo.png`, width: 1200, height: 630 }] },
    twitter: { card: 'summary_large_image', title, description, images: [`${SITE}/renterlogo.png`] },
  };
}

export default async function BlogListPage({ params }: { params: Promise<{ locale: 'ar'|'en'; hub?: string; n?: string }> }) {
  const { locale, hub, n } = await params;
  const page = Number(n ?? 1);
  const base = hub ? `/${locale}/blog/${hub}/` : `/${locale}/blog/`;
  if (n === '1') permanentRedirect(base);                           // /page/1/ -> 301 -> base
  if (!Number.isInteger(page) || page < 1) notFound();
  const h = hub ? await getHub(hub, locale) : null;
  if (hub && !h) notFound();
  const { items, meta } = await getPosts({ locale, hub, page, per_page: PER_PAGE });
  if (items.length === 0) notFound();                               // beyond last page -> 404
  const path = page > 1 ? `${base}page/${page}/` : base;

  const collection = {
    '@context': 'https://schema.org',
    '@type': 'CollectionPage',
    name: h ? h.h1 : (locale === 'ar' ? 'مدونة رينتر: أدلة وأسعار الإيجار في السعودية' : 'Renter Blog: Rental Guides and Prices in Saudi Arabia'),
    url: `${SITE}${path}`,
    inLanguage: locale,
    isPartOf: { '@id': `${SITE}/${locale}/#website` },
    mainEntity: { '@type': 'ItemList', itemListElement: items.map((p, i) => ({ '@type': 'ListItem', position: i + 1, url: `${SITE}/${locale}/blog/${p.hub}/${p.slug}/` })) },
  };
  const crumbs = [
    { '@type': 'ListItem', position: 1, name: locale === 'ar' ? 'الرئيسية' : 'Home', item: `${SITE}/${locale}/` },
    hub ? { '@type': 'ListItem', position: 2, name: locale === 'ar' ? 'المدونة' : 'Blog', item: `${SITE}/${locale}/blog/` } : { '@type': 'ListItem', position: 2, name: locale === 'ar' ? 'المدونة' : 'Blog' },
    ...(hub ? [{ '@type': 'ListItem', position: 3, name: h!.name }] : []),
  ];
  const breadcrumb = { '@context': 'https://schema.org', '@type': 'BreadcrumbList', itemListElement: crumbs };

  const pageLink = (p: number) => (p === 1 ? base : `${base}page/${p}/`);

  return (
    <main className="wrap">
      <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(collection).replace(/</g, '\\u003c') }} />
      <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(breadcrumb).replace(/</g, '\\u003c') }} />
      <h1>{h ? h.h1 : (locale === 'ar' ? 'مدونة رينتر: أدلة وأسعار الإيجار في السعودية' : 'Renter Blog: Rental Guides and Prices in Saudi Arabia')}{page > 1 && (locale === 'ar' ? `، الصفحة ${page}` : `, page ${page}`)}</h1>
      {h && <p className="hub-intro">{h.intro}</p>}
      {h && <a className="btn ghost" href={h.category_url} data-cta="category" data-cta-position="answer">{locale === 'ar' ? `تصفّح ${h.name} على رينتر` : `Browse ${h.name} on Renter`}</a>}
      <div className="cards">{items.map(p => <BlogCard key={p.slug} post={p} locale={locale} />)}</div>
      <nav className="pager" aria-label={locale === 'ar' ? 'الصفحات' : 'Pages'}>
        {page > 1 && <a href={pageLink(page - 1)} rel="prev">‹</a>}
        {Array.from({ length: meta.last_page }, (_, i) => i + 1).map(p => p === page ? <span key={p} aria-current="page">{p}</span> : <a key={p} href={pageLink(p)}>{p}</a>)}
        {page < meta.last_page && <a href={pageLink(page + 1)} rel="next">›</a>}
      </nav>
    </main>
  );
}
// rel="prev"/"next" on the <a> are harmless link hints; no <link rel="next"> in <head>: Google ignores it.

6. Sitemap and revalidation

Two new children in the sitemap index. lastmod is the content update time, not the render time. Paginated pages and author pages are not listed. No hreflang in the sitemap: the head already carries it and one of Google's three methods is enough.

// app/sitemap/[locale]/blog.xml/route.ts   ->  https://renter.net/sitemap/ar/blog.xml  and  /sitemap/en/blog.xml
import { getSitemapEntries } from '@/lib/blog/api';
const SITE = 'https://renter.net';
const HUBS = ['properties','cars-and-vehicles','occasions-and-entertainment','services','devices-and-appliances','equipment-and-tools','courts-and-arenas','furniture-and-accessories','fashion-and-clothes','guides'];

export const revalidate = 3600;

export async function GET(_: Request, { params }: { params: Promise<{ locale: 'ar'|'en' }> }) {
  const { locale } = await params;
  const posts = await getSitemapEntries(locale);          // [{ hub, slug, updated_at_content }], published only, en only if en_published
  const latest = posts[0]?.updated_at_content ?? new Date().toISOString();
  const urls = [
    { loc: `${SITE}/${locale}/blog/`, lastmod: latest },
    ...HUBS.map(h => ({ loc: `${SITE}/${locale}/blog/${h}/`, lastmod: posts.find(p => p.hub === h)?.updated_at_content ?? latest })),
    ...posts.map(p => ({ loc: `${SITE}/${locale}/blog/${p.hub}/${p.slug}/`, lastmod: p.updated_at_content })),
  ];
  const xml = `<?xml version="1.0" encoding="UTF-8"?>\n<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">\n` +
    urls.map(u => `  <url><loc>${u.loc}</loc><lastmod>${u.lastmod}</lastmod></url>`).join('\n') + `\n</urlset>`;
  return new Response(xml, { headers: { 'Content-Type': 'application/xml; charset=utf-8', 'Cache-Control': 'public, max-age=3600' } });
}

// Then add to the sitemap index (the file that serves https://renter.net/sitemap.xml):
//   <sitemap><loc>https://renter.net/sitemap/ar/blog.xml</loc></sitemap>
//   <sitemap><loc>https://renter.net/sitemap/en/blog.xml</loc></sitemap>

// app/api/revalidate/route.ts   (Laravel calls it after publish / update / unpublish)
import { revalidatePath } from 'next/cache';
import { NextRequest, NextResponse } from 'next/server';

export async function POST(req: NextRequest) {
  if (req.headers.get('x-revalidate-secret') !== process.env.BLOG_REVALIDATE_SECRET) return NextResponse.json({ ok: false }, { status: 401 });
  const { hub, slug } = await req.json();               // { "hub": "cars-and-vehicles", "slug": "car-rental-monthly-price-riyadh" }
  for (const locale of ['ar', 'en']) {
    revalidatePath(`/${locale}/blog/${hub}/${slug}`);
    revalidatePath(`/${locale}/blog/${hub}`);
    revalidatePath(`/${locale}/blog`);
    revalidatePath(`/sitemap/${locale}/blog.xml`);
  }
  return NextResponse.json({ ok: true });
}

// components/blog/ArticleView.tsx  ("use client"; renders nothing; one push per page view)
'use client';
import { useEffect } from 'react';
declare global { interface Window { dataLayer: Record<string, unknown>[] } }

export function ArticleView(p: { slug: string; hub: string; locale: string; author: string; format: string; wordCount: number }) {
  useEffect(() => {
    window.dataLayer = window.dataLayer || [];
    window.dataLayer.push({
      event: 'article_view',
      article_slug: p.slug, article_hub: p.hub, article_locale: p.locale,
      article_author: p.author, article_format: p.format, word_count: p.wordCount,
    });
    const onClick = (e: MouseEvent) => {
      const el = (e.target as HTMLElement).closest<HTMLElement>('[data-cta]');
      if (!el) return;
      window.dataLayer.push({
        event: 'article_cta_click',
        cta_type: el.dataset.cta,                       // category | product | app | related_article | hub
        cta_position: el.dataset.ctaPosition ?? 'body', // answer | mid | end | sidebar | body
        destination_url: (el as HTMLAnchorElement).href ?? '',
        article_slug: p.slug, article_hub: p.hub,
      });
    };
    document.addEventListener('click', onClick);
    return () => document.removeEventListener('click', onClick);
  }, [p.slug]);
  return null;
}
